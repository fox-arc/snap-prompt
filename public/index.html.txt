<script>
let uploadedImage = null;

const dropzone = document.getElementById('dropzone');
const fileInput = document.getElementById('fileInput');
const previewBox = document.getElementById('previewBox');
const previewImg = document.getElementById('previewImg');
const previewName = document.getElementById('previewName');
const removeBtn = document.getElementById('removeBtn');
const s2 = document.getElementById('s2');
const s3 = document.getElementById('s3');
const genBtn = document.getElementById('genBtn');
const errBox = document.getElementById('errBox');
const otherChip = document.getElementById('otherChip');
const otherInput = document.getElementById('otherInput');

dropzone.addEventListener('click', () => fileInput.click());
dropzone.addEventListener('dragover', e => { e.preventDefault(); dropzone.classList.add('dragover'); });
dropzone.addEventListener('dragleave', () => dropzone.classList.remove('dragover'));
dropzone.addEventListener('drop', e => {
  e.preventDefault();
  dropzone.classList.remove('dragover');
  if (e.dataTransfer.files.length) handleFile(e.dataTransfer.files[0]);
});
fileInput.addEventListener('change', e => {
  if (e.target.files.length) handleFile(e.target.files[0]);
});
removeBtn.addEventListener('click', () => {
  uploadedImage = null;
  previewBox.style.display = 'none';
  dropzone.style.display = 'block';
  fileInput.value = '';
  updateGateState();
});

function handleFile(file) {
  if (!file.type.startsWith('image/')) return;
  const reader = new FileReader();
  reader.onload = () => {
    const img = new Image();
    img.onload = () => {
      const MAX_DIM = 1280;
      let { width, height } = img;
      if (width > MAX_DIM || height > MAX_DIM) {
        const scale = MAX_DIM / Math.max(width, height);
        width = Math.round(width * scale);
        height = Math.round(height * scale);
      }
      const canvas = document.createElement('canvas');
      canvas.width = width;
      canvas.height = height;
      const ctx = canvas.getContext('2d');
      ctx.drawImage(img, 0, 0, width, height);
      const dataUrl = canvas.toDataURL('image/jpeg', 0.85);
      const [meta, b64] = dataUrl.split(',');
      const mediaType = meta.match(/data:(.*);base64/)[1];
      uploadedImage = { base64: b64, mediaType, name: file.name };
      previewImg.src = dataUrl;
      previewName.textContent = file.name;
      previewBox.style.display = 'flex';
      dropzone.style.display = 'none';
      s2.classList.remove('disabled');
      s3.classList.remove('disabled');
      updateGateState();
    };
    img.src = reader.result;
  };
  reader.readAsDataURL(file);
}

document.querySelectorAll('.chip').forEach(chip => {
  chip.addEventListener('click', () => {
    const cb = chip.querySelector('input');
    setTimeout(() => {
      chip.classList.toggle('active', cb.checked);
      if (chip === otherChip) {
        otherInput.classList.toggle('show', cb.checked);
        if (cb.checked) otherInput.focus();
      }
      updateGateState();
    }, 0);
  });
});
otherInput.addEventListener('input', updateGateState);

function setupRadio(el, input) {
  el.addEventListener('click', () => {
    input.checked = true;
    document.querySelectorAll('.radio-opt').forEach(o => o.classList.remove('active'));
    el.classList.add('active');
    updateGateState();
  });
}
const optOwn = document.getElementById('optOwn');
const optPhoto = document.getElementById('optPhoto');
setupRadio(optOwn, optOwn.querySelector('input'));
setupRadio(optPhoto, optPhoto.querySelector('input'));

function getSelectedAspects() {
  const list = [];
  document.querySelectorAll('#chipGrid input:checked').forEach(cb => {
    if (cb.value !== 'Lainnya') list.push(cb.value);
  });
  return list;
}
function getOtherText() {
  const cb = otherChip.querySelector('input');
  return cb.checked ? otherInput.value.trim() : '';
}
function getCharSrc() {
  const checked = document.querySelector('input[name="charsrc"]:checked');
  return checked ? checked.value : null;
}

function updateGateState() {
  const hasImage = !!uploadedImage;
  const hasAspect = getSelectedAspects().length > 0 || getOtherText().length > 0;
  const hasCharSrc = !!getCharSrc();
  genBtn.disabled = !(hasImage && hasAspect && hasCharSrc);
}

const resultCard = document.getElementById('resultCard');
const loadingBox = document.getElementById('loadingBox');
const resultText = document.getElementById('resultText');
const resultActions = document.getElementById('resultActions');
const copyBtn = document.getElementById('copyBtn');
const regenBtn = document.getElementById('regenBtn');

genBtn.addEventListener('click', generatePrompt);
regenBtn.addEventListener('click', generatePrompt);

async function generatePrompt() {
  errBox.style.display = 'none';
  resultCard.style.display = 'block';
  loadingBox.style.display = 'flex';
  resultText.style.display = 'none';
  resultActions.style.display = 'none';
  resultCard.scrollIntoView({ behavior: 'smooth', block: 'nearest' });

  const aspects = getSelectedAspects();
  const otherText = getOtherText();
  const charSrc = getCharSrc();

  let focusLine = aspects.length ? aspects.join(', ') : '';
  if (otherText) focusLine += (focusLine ? ', ' : '') + otherText;

  let instruction = `Kamu akan menerima 1 foto referensi. Tugasmu HANYA SATU: deskripsikan foto ini dengan sangat detail dalam bentuk prompt siap pakai untuk image generation model bernama GPT Image 2.\n\nTulis HANYA dalam Bahasa Indonesia. Jangan pakai kalimat pembuka, penjelasan tambahan, atau basa-basi apapun — langsung tulis isi promptnya saja.\n\nFokuskan deskripsi secara sangat detail pada aspek berikut: ${focusLine}.\n\n`;

  if (charSrc === 'own') {
    instruction += `PENTING: User akan attach karakter dan outfit miliknya sendiri secara terpisah saat generate gambar nanti. JANGAN mendeskripsikan wajah, identitas, bentuk tubuh, atau detail outfit dari foto ini sama sekali, walaupun itu termasuk salah satu fokus di atas.\n\nSebagai gantinya, buka prompt dengan kalimat instruksi kunci (lock instruction) kurang lebih seperti ini: "Gunakan karakter dan outfit yang telah di-attach/direferensikan secara terpisah. Pertahankan identitas wajah, bentuk tubuh, gaya rambut, dan seluruh detail outfit tersebut 100% tanpa perubahan sedikit pun."\n\nSetelah kalimat lock itu, lanjutkan dengan deskripsi sangat detail untuk aspek-aspek lain yang dipilih di atas (selain karakter/outfit), seperti pose, background, lighting, lensa kamera, atau vibes — semuanya tetap deskripsikan senyata dan sedetail mungkin berdasarkan foto.`;
  } else {
    instruction += `Deskripsikan juga karakter dan outfit yang terlihat di foto secara sangat detail (bentuk tubuh, gaya rambut, warna kulit, jenis dan warna pakaian, tekstur bahan, aksesoris, sepatu, dll) sebagai bagian dari prompt, sesuai fokus yang dipilih di atas.`;
  }

  try {
    const response = await fetch("/api/generate", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        image: uploadedImage.base64,
        mediaType: uploadedImage.mediaType,
        instruction: instruction
      })
    });

    const data = await response.json();
    if (!response.ok) throw new Error(data.error || "Request gagal (" + response.status + ")");

    const textParts = (data.content || []).filter(c => c.type === 'text').map(c => c.text);
    const finalText = textParts.join('\n').trim();

    if (!finalText) throw new Error("Responsnya kosong, coba generate ulang ya.");

    loadingBox.style.display = 'none';
    resultText.textContent = finalText;
    resultText.style.display = 'block';
    resultActions.style.display = 'flex';
  } catch (err) {
    console.error("Snap-Prompt error:", err);
    loadingBox.style.display = 'none';
    errBox.textContent = "Gagal generate: " + err.message + ". Coba lagi, atau pakai foto dengan ukuran lebih kecil.";
    errBox.style.display = 'block';
  }
}

copyBtn.addEventListener('click', () => {
  navigator.clipboard.writeText(resultText.textContent).then(() => {
    const original = copyBtn.textContent;
    copyBtn.textContent = "✅ Ke-copy!";
    setTimeout(() => copyBtn.textContent = original, 1500);
  });
});
</script>
